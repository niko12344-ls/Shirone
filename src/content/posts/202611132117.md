> 我家里正好有一个垫桌脚的随身WIFI 之前做过很多项目 最后吃灰了 但是随着许多网站开始推荐用户开启二次验证 就出现了这篇文章

由于这是一篇静态博客 需要在GitHub上编辑文章 并且GitHub需要强制绑定2fa 导致我在学校里很难编辑 所以我将这个板子改造成了一个totp 验证器
需要准备的材料
1. 一个随身WIFI
2. 一个会折腾的你

## 第2步开始进行正式操作
先登录这块板子的ssh
`ssh root@192.168.68.1`
然后安装`oathtool`
`sudo apt install oathtool`
通过任意文本编辑器编辑一个Shell脚本
vim totptool.sh
```bash
#!/bin/bash
KEYFILE="$HOME/.totp_keys"

check_file() {
    if [ ! -s "$KEYFILE" ]; then
        echo "密钥文件为空或不存在: $KEYFILE" >&2
        exit 1
    fi
}

# 生成一行：账号 验证码 剩余秒数
gen_line() {
    local name="$1" key="$2"
    local code
    code=$(oathtool --totp --base32 "$key" 2>/dev/null)
    local remain=$(( 30 - $(date +%s) % 30 ))
    if [ -z "$code" ]; then
        printf '%-20s %-8s %s\n' "$name" "失败" "检查密钥格式"
    else
        printf '%-20s %-8s %2ds\n' "$name" "$code" "$remain"
    fi
}

print_header() {
    printf '%-20s %-8s %s\n' "账号" "验证码" "剩余"
    printf -- '--------------------------------------------\n'
}

# 一次性列出全部
list_all() {
    check_file
    print_header
    while IFS=$'\t' read -r name key || [ -n "$name" ]; do
        [ -z "$name" ] && continue
        [[ "$name" == \#* ]] && continue
        gen_line "$name" "$key"
    done < "$KEYFILE"
}

# 实时刷新：每秒重绘，剩余秒数跳动
live_mode() {
    check_file
    clear
    while true; do
        printf '\033[H'          # 光标回左上角
        print_header
        while IFS=$'\t' read -r name key || [ -n "$name" ]; do
            [ -z "$name" ] && continue
            [[ "$name" == \#* ]] && continue
            gen_line "$name" "$key"
        done < "$KEYFILE"
        printf '\033[J'          # 清除下方残留
        sleep 1
    done
}

case "$1" in
    ""|all)
        list_all
        ;;
    --live)
        live_mode
        ;;
    *)
        check_file
        KEY=$(awk -F'\t' -v n="$1" '$1==n {print $2; exit}' "$KEYFILE")
        if [ -z "$KEY" ]; then
            echo "未找到: $1" >&2
            exit 1
        fi
        gen_line "$1" "$KEY"
        ;;
esac
```
然后保存退出 :wq
然后在目录下创建一个存储密钥的地方 
vim .totp_keys
密钥内容示例
```text
# 账号名	Base32密钥
cloudflare.com	MFRGGZDFMZTWQ2LKNNWG23TP
github.com	JBSWY3DPEHPK3PXP
google.com	GEZDGNBVGY3TQOJQ
aws.amazon.com	KRSXG5CTMVRXEZLU
# 下面这行是注释，会被跳过
# old-account	AAAAAAAAAAAAAAAA
```
建议赋予600权限只读
然后执行 ./totptool.sh
就可以看见验证码和剩余时间了
### 重要提醒
==如果你觉得这些密钥泄露的话 赶快重置 并且重新获取动态验证码 并且建议获取恢复码保存在安全的地方=={.error}
